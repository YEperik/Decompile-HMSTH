cmake_minimum_required(VERSION 3.16)
project(harvestmoon_native_fragments LANGUAGES CXX)
add_library(hm_native_fragments SHARED src/math_bits.cpp)
target_include_directories(hm_native_fragments PUBLIC include)
target_compile_features(hm_native_fragments PUBLIC cxx_std_17)
set_target_properties(hm_native_fragments PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(hm_native_fragments PRIVATE /W4 /WX)
else()
  target_compile_options(hm_native_fragments PRIVATE -Wall -Wextra -Werror -Wconversion)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(hm_display_test tests/display_integration.cpp)
  target_link_libraries(hm_display_test PRIVATE hm_native_fragments)
  if(MSVC)
    target_compile_options(hm_display_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(hm_display_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME native_display_integration COMMAND hm_display_test)
endif()
